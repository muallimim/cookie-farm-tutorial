3. 7/24 Otonom Telegram Satış Botu Mimarisi

Yazılımcılar ve bot işletmecileri gece çalışır. Bir web sitesine girip üye olmakla vakit kaybetmek istemezler. Bir Telegram botu açıp /bakiye_yukle ve /satin_al diyerek saniyeler içinde çerez almak isterler.

🤖 Python aiogram ile Geliştirilmiş Mini Satış Botu Kodu
import asyncio
from aiogram import Bot, Dispatcher, types
from aiogram.filters import Command

API_TOKEN = "TELEGRAM_BOT_TOKEN_BURAYA"
bot = Bot(token=API_TOKEN)
dp = Dispatcher()

PACKAGES = {
    "pack_500": {"name": "500 Adet Organik Çerez", "price_usdt": 5.0, "file": "packages/batch_500.zip"},
    "pack_1000": {"name": "1.000 Adet Organik Çerez", "price_usdt": 10.0, "file": "packages/batch_1000.zip"}
}

@dp.message(Command("start"))
async def start_handler(msg: types.Message):
    await msg.reply(
        "👋 Hoş Geldiniz! Endüstriyel Organik Çerez Botu.\n\n"
        "📦 Güncel Stok: 45.000 Adet Taze TR/US Çerez\n"
        "🚀 Format: Dolphin{anty}, AdsPower ve JSON\n\n"
        "Paketler için /paketler komutunu kullanabilirsiniz."
    )

@dp.message(Command("paketler"))
async def packages_handler(msg: types.Message):
    text = "🛒 SATIN ALINABİLİR PAKETLER:\n\n"
    for k, v in PACKAGES.items():
        text += f"🔹 {v['name']} — {v['price_usdt']}$ (USDT TRC20)\n"
    text += "\nÖdeme ve anında dosya teslimi için: /satin_al "
    await msg.reply(text)

@dp.message(Command("satin_al"))
async def buy_handler(msg: types.Message):
    # Kripto onay mekanizması sonrası anında dosya fırlat:
    zip_file = types.FSInputFile("packages/batch_1000.zip")
    await msg.reply_document(zip_file, caption="✅ Ödeme Onaylandı! 1.000 Adet Çerez Paketiniz Ektedir.")

if __name__ == "__main__":
    dp.run_polling(bot)
